SUBROUTINE validTransformation(value, type)
    contains_name ← FALSE
    
    //Only parses value if not boolean
    IF type <> "reflect" THEN
        tokens = tokenize(value) 
        
        //If NAME pattern in value, will be rejected
        FOR token IN tokens 
            IF token.kind = "NAME" THEN
                contains_name ← TRUE
                RETURN FALSE
            ENDIF
        ENDFOR
    ENDIF

    // Checking if the value is default, if so the transformation should be skipped
    IF NOT contains_name THEN
        IF type = "shift" AND EVALUATE(value) = 0 THEN
            RETURN FALSE
        ELSE IF type = "stretch" AND EVALUATE(value) = 1 THEN
            RETURN FALSE
        ELSE IF type = "reflect" AND value = FALSE THEN
            RETURN FALSE
        ELSE
            RETURN TRUE
        ENDIF
    ENDIF

    RETURN FALSE
ENDSUBROUTINE

SUBROUTINE enqueueTransformations(stretchX, stretchY, reflectX, reflectY, shiftX, shiftY)
    // Get values from each entry box
    stretchX_value ← stretchX.getText() 
    stretchY_value ← stretchY.getText() 
    reflectX_bool ← reflectX.IS_SELECTED()
    reflectY_bool ← reflectY.IS_SELECTED()
    shiftX_value ← shiftX.getText()
    shiftY_value ← shiftY.getText()

    // Adds each transformation to the queue in the correct order if validTransformation returns true
    // Uses EVALUATE to convert an invalid entry such as 3 + 2 to 5
    // Needs to replace all ^ with ** as this is done in the parser which is unused here
    IF validTransformation(stretchX_value, "stretch") THEN 
        queue.ENQUEUE("stretch", EVALUATE(REPLACE_ALL(stretchX_value, "^", "**")), 'x')
    ENDIF

    IF validTransformation(stretchY_value, "stretch") THEN 
        queue.ENQUEUE("stretch", EVALUATE(REPLACE_ALL(stretchY_value, "^", "**")), 'y')
    ENDIF

    IF validTransformation(reflectX_bool, "reflect") THEN 
        queue.ENQUEUE("reflect", reflectX_bool, 'x')
    ENDIF

    IF validTransformation(reflectY_bool, "reflect") THEN 
        queue.ENQUEUE("reflect", reflectY_bool, 'y')
    ENDIF

    IF validTransformation(shiftX_value, "shift") THEN 
        queue.ENQUEUE("shift", EVALUATE(REPLACE_ALL(shiftX_value, "^", "**")), 'x')
    ENDIF

    IF validTransformation(shiftY_value, "shift") THEN 
        queue.ENQUEUE("shift", EVALUATE(REPLACE_ALL(shiftY_value, "^", "**")), 'y')
    ENDIF
ENDSUBROUTINE
