CLASS FunctionEntry
    ATTRIBUTES
        PRIVATE user_input : STRING
        PRIVATE tokens : LIST
        PRIVAVTE variable : STRING
        PRIVATE ast : ASTNODE OBJECT OR NULL
    ENDATTRIBUTES
    
    CONSTRUCTOR (user_input)
        user_input ← user_input
        tokens ← EMPTY LIST
        variable ← EMPTY STRING
        ast ← NULL
    ENDCONSTRUCTOR

    PUBLIC METHOD PARSE_FUNCTION()
        // Tokenise the user input string
        tokens, variable ← PARSE(user_input)

        IF tokens IS EMPTY THEN
            RETURN FALSE
        ENDIF

        RETURN TRUE
    ENDMETHOD

    PUBLIC METHOD FUNCTION_AST()
        // Convert tokens to postfix then build the AST
        IF tokens IS EMPTY THEN
            RETURN
        ENDIF

        tokens ← POSTFIX(tokens)
        ast ← POSTFIX_TO_AST(tokens)
    ENDMETHOD

    PUBLIC METHOD OUTPUT_FUNCTION()
        // Return the AST
        IF ast = NULL THEN
            RETURN NULL
        ELSE
            OUTPUT "Function Variable:", variable
            RETURN ast, variable
        ENDIF
    ENDMETHOD
ENDCLASS
