CLASS  MODIFY_FUNCTION
    ATTRIBUTES
        PRIVATE altered_function : FUNCTION OBJECT
    ENDATTRIBUTES

    CONSTSRUCTOR (altered_function)
        // Store the function that will be modified
        altered_function ← altered_function
    ENDCONSTRUCTOR

    PUBLIC METHOD  MODIFY_FUNCTION()
        // Base method for all function modifications
        // Just returns the function unchanged
        RETURN altered_function
    ENDMETHOD
ENDCLASS

CLASS DifferentiateFunction INHERITS  MODIFY_FUNCTION
    PUBLIC METHOD  MODIFY_FUNCTION()
        // Applies differentiation to the function AST

        new_tree ← DIFFERENTIATE(altered_function.GET_FUNCTION())

        IF new_tree = NONE THEN
            RETURN NONE
        ENDIF

        // Simplifies the AST 
        simplified ← SIMPLIFY_AST(new_tree)

        altered_function.SET_FUNCTION(simplified)

        RETURN altered_function
    ENDMETHOD
ENDCLASS

CLASS ShiftFunction INHERITS  MODIFY_FUNCTION
    ATTRIBUTES
        PRIVATE altered_function : FUNCTION OBJECT
        PRIVATE axis : STRING
        PRIVATE value : FLOAT
    ENDATTRIBUTES
    
    CONSTRUCTOR (altered_function, axis, value)
        // Store the function to be modified
        altered_function ← altered_function

        // Store axis and shift value
        axis ← axis
        value ← value
    ENDCONSTRUCTOR
    
    PUBLIC METHOD  MODIFY_FUNCTION()
        // Applies a shift to the function AST

        tree ← altered_function.GET_FUNCTION()

        IF axis = "x" THEN
            tree ← SHIFT_X(tree, value)
        ELSE IF axis = "y" THEN
            tree ← SHIFT_Y(tree, value)
        ENDIF

        altered_function.SET_FUNCTION(tree)

        RETURN altered_function
    ENDMETHOD
ENDCLASS

CLASS ReflectFunction INHERITS  MODIFY_FUNCTION
    CONSTRUCTOR (altered_function, axis)
        // Store the function to be modified
        altered_function ← altered_function

        // Store the axis of reflection
        axis ← axis
    ENDCONSTRUCTOR

    PUBLIC METHOD  MODIFY_FUNCTION()
        // Applies a reflection to the function AST

        tree ← altered_function.GET_FUNCTION()

        IF axis = "x" THEN
            tree ← REFLECT_X_AXIS(tree)
        ELSE IF axis = "y" THEN
            tree ← REFLECT_Y_AXIS(tree)
        ENDIF

        altered_function.SET_FUNCTION(tree)

        RETURN altered_function
    ENDMETHOD
