CLASS TransformManager
    // Manages a function and a queue of transformations
    // Responsible for applying transformations in the correct order

    SUBROUTINE Initialise(input_expression)
        // Queue storing transformations to be applied
        transformations_queue ← new Queue(6)

        // Original base function
        original_function ← input_expression

        // Function currently being transformed
        current_function ← input_expression
    ENDSUBROUTINE

    SUBROUTINE addTransformations(x_stretch_box, y_stretch_box, x_shift_box, y_shift_box, x_reflect, y_reflect)
        // Adds given transformations to the queue
        transformations_queue ← enqueueTransformations(
            x_stretch_box,
            y_stretch_box,
            x_shift_box,
            y_shift_box,
            x_reflect,
            y_reflect
        )
    ENDSUBROUTINE

    SUBROUTINE applyTransformation(transformation, update_base)
        // Applies a single transformation to the current function

        modifier ← NONE

        IF transformation.type = "shift" THEN
            modifier ← new ShiftFunction(current_function, transformation.axis, transformation.value)

        ELSE IF transformation.type = "stretch" THEN
            modifier ← new StretchFunction(current_function, transformation.axis, transformation.value)

        ELSE IF transformation.type = "reflect" THEN
            modifier ← new ReflectFunction(current_function, transformation.axis)

        ELSE IF transformation.type = "differentiate" THEN
            modifier ← new DifferentiateFunction(current_function)
        ENDIF

        IF modifier ≠ NONE THEN
            new_function ← modifier.ModifyFunction()

            IF update_base = TRUE THEN
                current_function ← new_function
            ENDIF

            RETURN new_function
        ENDIF

        RETURN NONE
    ENDSUBROUTINE

    SUBROUTINE applyAllTransformations(graph_plotter)
        // Applies every transformation in the queue sequentially

        WHILE transformations_queue.isEmpty() = FALSE
            transformation ← transformations_queue.dequeue()
            new_function ← applyTransformation(transformation, TRUE)
            graph_plotter.plotSubsequent(new_function)
        ENDWHILE

        // Reset queue after all transformations applied
        transformations_queue ← new Queue(6)
    ENDSUBROUTINE

    SUBROUTINE getCurrentFunction()
        // Returns the current function
        RETURN current_function
    ENDSUBROUTINE

    SUBROUTINE nextTransformation()
        // Applies the next transformation only

        IF transformations_queue.isEmpty() THEN
            RETURN NONE
        ENDIF

        // Get next transformation
        transformation ← transformations_queue.dequeue()

        // Create copy of current function
        ast_copy ← copyAST(current_function.getFunction())
        var_copy ← current_function.getVariable()
        new_function ← new Function(ast_copy, var_copy)

        modifier ← NONE

        IF transformation.getType() = "shift" THEN
            modifier ← new ShiftFunction(new_function, transformation.getAxis(), transformation.getVal())

        ELSE IF transformation.getType() = "stretch" THEN
            modifier ← new StretchFunction(new_function, transformation.getAxis(), transformation.getVal())

        ELSE IF transformation.getType() = "reflect" THEN
            modifier ← new ReflectFunction(new_function, transformation.getAxis())

        ELSE IF transformation.getType() = "differentiate" THEN
            modifier ← new DifferentiateFunction(new_function)
        ENDIF

        IF modifier ≠ NONE THEN
            current_function ← modifier.ModifyFunction()
            RETURN current_function
        ENDIF

        RETURN NONE
    ENDSUBROUTINE

    SUBROUTINE setBaseFunction(new_function)
        // Sets a new base function for repeated transformations
        current_function ← new_function
        transformations_queue ← new Queue(1)
    ENDSUBROUTINE

    SUBROUTINE hasTransformations()
        // Returns TRUE if transformations remain in the queue
        RETURN transformations_queue.isEmpty() = FALSE
    ENDSUBROUTINE
ENDCLASS
