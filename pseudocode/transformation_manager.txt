CLASS TRASNFORM_MANAGER
    // Manages a function and a queue of transformations
    // Responsible for applying transformations in the correct order

    ATTRIBUTES
        PRIVATE transformations_queue : QUEUE OBJECT
        PRIVATE current_function : FUNCTION OBJECT
    ENDATTRIBUTES

    CONSTRUCTOR (input_expression)
        // Queue storing transformations to be applied
        transformations_queue ← NEW QUEUE(6)
        // Function currently being transformed
        current_function ← input_expression
    ENDCONSTRUCTOR

    PUBLIC METHOD ADD_TRANSFORMATIONS(x_stretch_box, y_stretch_box, x_shift_box, y_shift_box, x_reflect, y_reflect)
        // Adds given transformations to the queue
        transformations_queue.ENQUEUE(x_stretch_box)
        transformations_queue.ENQUEUE(Y_stretch_box)
        transformations_queue.ENQUEUE(x_shift_box)
        transformations_queue.ENQUEUE(Y_shift_box)
        transformations_queue.ENQUEUE(x_reflect)
        transformations_queue.ENQUEUE(Y_reflect)
    ENDMETHOD

    PUBLIC METHOD APPLY_TRANSFORMATION(transformation)
        // Applies a single transformation to the current function

        new_function ← NULL

        IF transformation.TYPE = "shift" THEN
            new_function ← NEW SHIFT_FUNCTION(current_function, transformation.axis, transformation.value)

        ELSE IF transformation.TYPE = "stretch" THEN
            new_function ← NEW STRETCH_FUNCTION(current_function, transformation.axis, transformation.value)

        ELSE IF transformation.TYPE = "reflect" THEN
            new_function ← NEW REFLECT_FUNCTION(current_function, transformation.axis)

        ELSE IF transformation.TYPE = "differentiate" THEN
            new_function ← NEW DIFFERENTIATE_FUNCTION(current_function)
        ENDIF

        IF new_function <> NULL THEN
            RETURN new_function
        ENDIF

        RETURN NULL
    ENDMETHOD

    PUBLIC METHOD APPLY_ALL_TRANSFORMATIONS(graph_plotter)
        // Applies every transformation in the queue sequentially

        WHILE transformations_queue.IS_EMPTY() = FALSE
            transformation ← transformations_queue.DEQUEUE()
            new_function ← APPLY_TRANSFORMATION(transformation, TRUE)
            graph_plotter.PLOT_SUBSEQUENT(new_function)
        ENDWHILE

        // Reset queue after all transformations applied
        transformations_queue ← NEW QUEUE(6)
    ENDMETHOD

    PUBLIC METHOD GET_CURRENT_FUNCTION()
        // Returns the current function
        RETURN current_function
    ENDMETHOD

    PUBLIC METHOD SET_BASE_FUNCTION(new_function)
        // Sets a new base function for repeated transformations
        current_function ← new_function
    ENDMETHOD

    PUBLIC METHOD HAS_TRANSFORMATIONS()
        // Returns TRUE if transformations remain in the queue
        RETURN transformations_queue.IS_EMPTY() = FALSE
    ENDMETHOD
ENDCLASS
