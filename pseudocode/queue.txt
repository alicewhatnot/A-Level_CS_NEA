CLASS QUEUE
    ATTRIBUTES
        PRIVATE items : LIST
        PRIVATE front : INTEGER
        PRIVATE rear : INTEGER
        PRIVATE size : INTEGER
        PRIVATE max_size : INTEGER
    ENDATTRIBUTES

    CONSTRUCTOR (max_size)
        // Create fixed-size array for queue items
        items ← EMPTY LIST

        // Index of the front of the queue
        front ← 0

        // Index of the rear of the queue
        rear ← -1

        // Current number of items in the queue
        size ← 0

        // Maximum size of the queue
        max_size ← max_size
    ENDCONSTRUCTOR

    PUBLIC METHOD ENQUEUE(value)
        // Adds an item to the rear of the queue

        IF IS_FULL() THEN
            RETURN
        ENDIF

        rear ← (rear + 1) MOD max_size
        items[rear] ← value
        size ← size + 1
    ENDMETHOD

    PUBLIC METHOD DEQUEUE()
        // Removes and returns the item at the front of the queue

        IF IS_EMPTY() THEN
            RETURN NONE
        ENDIF

        item ← items[front]
        items[front] ← NONE

        front ← (front + 1) MOD max_size
        size ← size - 1

        RETURN item
    ENDMETHOD

    PUBLIC METHOD IS_EMPTY()
        // Returns TRUE if the queue is empty
        RETURN size = 0
    ENDMETHOD

    PUBLIC METHOD IS_FULL()
        // Returns TRUE if the queue is full
        RETURN size = max_size
    ENDSUBROUTINE

    PUBLIC METHOD CLEAR()   
        // Removes all items from the queue

        WHILE IS_EMPTY() = FALSE
            DEQUEUE()
        ENDWHILE
    ENDMETHOD
ENDCLASS
