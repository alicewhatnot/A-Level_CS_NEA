SUBROUTINE DIFFERENTIATE(node)
    // Recursively DIFFERENTIATEs a node and its relevant children
    IF node = NULL THEN
        RETURN NULL

    // Returns the derivative of a number which is 0
    ELSE IF node.TYPE = "NUMBER" THEN
        RETURN NEW ASTNODE("NUMBER", "0", NULL, NULL)

    // Returns the derivative of a single variable which is 1
    ELSE IF node.TYPE = "NAME" THEN
        RETURN NEW ASTNODE("NUMBER", "1", NULL, NULL)

    ELSE IF node.TYPE = "OP" THEN
        operator ← node.VALUE

        // differentiates left and right of the operator, using the rule d/dx[f ± g] = f' ± g'
        IF operator = "+" OR operator = "-" THEN
            left_differentiated ← DIFFERENTIATE(node.LEFT) 
            right_differentiated ← DIFFERENTIATE(node.RIGHT) 
            RETURN NEW ASTNODE("OP", operator, left_differentiated, right_differentiated)

        ELSE IF operator = "*" THEN
        // Implements the product rule (d/dx[fg] = f'g + fg') to differentiate left * right
            left ← node.LEFT
            right ← node.RIGHT
            left_differentiated ← DIFFERENTIATE(node.LEFT)
            right_differentiated ← DIFFERENTIATE(node.RIGHT)

            term1 ← NEW ASTNODE("OP", "*", left_differentiated, right)
            term2 ← NEW ASTNODE("OP", "*", right_differentiated, left)
            RETURN NEW ASTNODE("OP", "+", term1, term2)

        ELSE IF operator = "**" THEN
        // Implements the power rule (d/dx[u^n] = n * u^(n-1) * u') to differentiate a base to a numerical exponent
            base ← node.LEFT
            exponent ← node.RIGHT

            IF exponent.type = "NUMBER" THEN
                new_exponent ← NEW ASTNODE("NUMBER", STRING(INT(exponent.value) - 1), NULL, NULL)
                base_differentiated ← DIFFERENTIATE(base)

                value_to_power ← NEW ASTNODE("OP", "**", base, new_exponent)
                coefficient ← NEW ASTNODE("NUMBER", exponent.VALUE, NULL, NULL)
                product ← NEW ASTNODE("OP", "*", coefficient, value_to_power)
                RETURN NEW ASTNODE("OP", "*", product, base_differentiated)
            
            ELSE IF exponent.type = "NAME" AND base.type = "NUMBER" THEN
                // Implements the rule d/dx[a^x] = a^x * ln(a) to differentiate a numerical base to a variable exponent                
                ln_base ← NEW ASTNODE("FUNCTION", "ln", NEW ASTNODE("NUMBER", base.value, NULL, NULL), NULL)
                value_to_power ← NEW ASTNODE("OP", "**", NEW ASTNODE("NUMBER", base.value, NULL, NULL), exponent)
                RETURN NEW ASTNODE("OP", "*", value_to_power, ln_base)

            ELSE
                RETURN NULL
            ENDIF
        ENDIF
    
    // differentiates a function using the chain rule (d/dx[f(g)] = f'g')
    ELSE IF node.TYPE = "FUNCTION" THEN
        // If argument is a constant number, derivative is 0
        // Implemented for further derivatives of exponential functions
        IF node.LEFT.TYPE == "NUMBER" THEN
            RETURN NEW ASTNODE("NUMBER", "0", NULL, NULL)

        function ← node.VALUE
        argument ← node.LEFT
        argument_differentiated ← DIFFERENTIATE(argument)

        // Uses the chain rule for the cases of sin and cos 
        IF function = "sin" THEN
            cos_node ← NEW ASTNODE("FUNCTION", "cos", argument, NULL)
            RETURN NEW ASTNODE("OP", "*", cos_node, argument_differentiated)

        ELSE IF function = "cos" THEN
            sin_node ← NEW ASTNODE("FUNCTION", "sin", argument, NULL)
            negative ← NEW ASTNODE("NUMBER", "-1", NULL, NULL)
            negative_sin ← NEW ASTNODE("OP", "*", sin_node, negative)
            RETURN NEW ASTNODE("OP", "*", negative_sin, argument_differentiated)

        // Splits tan into sin/cos to differentiate
        ELSE IF function = "tan" THEN
            cos_node ← NEW ASTNODE("FUNCTION", "cos", argument, NULL)
            cos_squared ← NEW ASTNODE("OP", "*", cos_node, cos_node)
            one NEW ← ASTNODE("NUMBER", "1", NULL, NULL)
            reciprocal ← NEW ASTNODE("OP", "/", one, cos_squared)

            RETURN NEW ASTNODE("OP", "*", reciprocal, argument_differentiated)

        ENDIF
    ENDIF

    // Safety catch if the node cannot be differentiated
    RETURN NULL
ENDSUBROUTINE
