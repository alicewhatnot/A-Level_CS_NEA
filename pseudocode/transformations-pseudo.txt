SUBROUTINE reflectXAxis(node)
    // Performs a reflection in the x axis
    IF node = NULL THEN
        RETURN NULL

    // Create new nodes at root to multiply expression by -1
    negative_one ← NEW ASTNode("NUMBER", "-1", NULL, NULL)
    RETURN NEW ASTNode("OP", "*", negative_one, node)
ENDSUBROUTINE

SUBROUTINE reflectYAxis(node)
    // Performs a reflection in the y axis
    IF node = NULL THEN
        RETURN NULL

    // Replaces every x with -1 * x in the AST
    ELSE IF node.type = "NAME" AND node.value = "x" THEN
        negative_one ← NEW ASTNode("NUMBER", "-1", NULL, NULL)
        RETURN NEW ASTNode("OP", "*", negative_one, node)

    ELSE
        node.left ← reflectYAxis(node.left)
        node.right ← reflectYAxis(node.right)
        RETURN node
    ENDIF
ENDSUBROUTINE

SUBROUTINE shiftX(node, shift)
    // Performs a shift along the x axis
    IF node = NULL THEN
        RETURN NULL

    // Replace x with x - shift so a positive shift moves right
    ELSE IF node.type = "NAME" AND node.value = "x" THEN
        shift_amount ← NEW ASTNode("NUMBER", STRING(shift), NULL, NULL)
        RETURN NEW ASTNode("OP", "-", node, shift_amount)

    ELSE
        node.left ← shiftX(node.left, shift)
        node.right ← shiftX(node.right, shift)
        RETURN node
    ENDIF
ENDSUBROUTINE

SUBROUTINE shiftY(node, shift)
    // Performs a shift along the y axis
    IF node = NULL THEN
        RETURN NULL

    // Adds the shift to the expression 
    shift_amount ← NEW ASTNode("NUMBER", STRING(shift), NULL, NULL)
    RETURN NEW ASTNode("OP", "+", node, shift_amount)
ENDSUBROUTINE

SUBROUTINE stretchX(node, stretch)
    // Performs a stretch along the x axis
    IF node = NULL THEN
        RETURN NULL

    // Divide by stretch scale factor 
    ELSE IF node.type = "NAME" AND node.value = "x" THEN
        factor ← NEW ASTNode("NUMBER", STRING(stretch), NULL, NULL)
        RETURN NEW ASTNode("OP", "/", node, factor)

    ELSE
        node.left ← stretchX(node.left, stretch)
        node.right ← stretchX(node.right, stretch)
        RETURN node
    ENDIF
ENDSUBROUTINE

SUBROUTINE stretchY(node, stretch)
    // Performs a stretch along the y axis
    IF node = NULL THEN
        RETURN NULL

    // Simply multiply the expression AST by stretch scale factor
    factor ← NEW ASTNode("NUMBER", STRING(stretch), NULL, NULL)
    RETURN NEW ASTNode("OP", "*", factor, node)
ENDSUBROUTINE
