SUBROUTINE tokenize(expression)
    // Initialise variables and clean the expression
    tokens ← []
    index ← 0
    valid_characters ← TRUE
    REMOVE all spaces from expression
    REPLACE_ALL(expression, "^", "**")

    // Regex patterns to look for in the expression
    patterns ← (
    NUMBER: digit followed by optional . followed by digit OR
    FUNCTION: one of 'sin', 'cos', 'tan' OR
    NAME: any single letter from a–z or A–Z OR
    OP: one of '+', '-', '*', '/', '**' OR
    LEFTPARENTHESIS: '(' OR
    RIGHTPARENTHESIS: ')'
    )

 
    // Loop through the expression, adding characters and corresponding token to tokens list
    WHILE index < LEN(expression) AND validCharacters = TRUE
        matched ← FALSE

        IF expression[index TO ...] matches NUMBER pattern THEN
            token ← "NUMBER"
            characters ← matched substring
            matched ← TRUE

        ELSE IF expression[index TO ...] matches FUNCTION pattern THEN
            token ← "FUNCTION"
            characters ← matched substring
            matched ← TRUE

        ELSE IF expression[index] matches NAME pattern THEN
            token ← "NAME"
            characters ← expression[index]
            matched ← TRUE

        // Checking for operators must take into account that only one operator is double length
        ELSE IF expression[index] matches OP pattern THEN
            IF expression[index] = * AND expression[index + 1] = * THEN
                characters ← '**'
            token ← "OP"
            characters ← expression[index]
            matched ← TRUE

        ELSE IF expression[index] matches LEFTPARENTHESIS pattern THEN
            token ← "LEFTPARENTHESIS"
            characters ← expression[index]
            matched ← TRUE

        ELSE IF expression[index] matches RIGHTPARENTHESIS pattern THEN
            token ← "RIGHTPARENTHESIS"
            characters ← expression[index]
            matched ← TRUE

        ELSE
            valid_characters ← FALSE
        ENDIF
    
        // Add the token and characters to the list, incrementing index according to amount of characters
        IF matched THEN
            tokens.APPEND((token, characters))
            index ← index + LEN(characters)
    ENDWHILE

    // Only returns token list if all characters are valid
    IF valid_characters THEN
        RETURN tokens
    ELSE
        RETURN NULL
    ENDIF
ENDSUBROUTINE

SUBROUTINE insertImplicitMultiplication(tokens)
    // Adds multiplication where the user would consider it to be implicit
    // Initialise variables
    new_tokens ← []
    length ← LEN(tokens)

    FOR index ← 1 TO length - 1
        current_token ← tokens[index, 1]
        current_value ← tokens[index, 2]
        next_token ← tokens[index + 1, 1]

        new_tokens.APPEND(current_token, current_value)

        // If two relevant tokens are next to each other, insert a new ‘*’ operator token
        IF (current_token IN ["NUMBER", "NAME", "RIGHTPARENTHESIS"] AND  next_token IN ["NAME", "FUNCTION", "LEFTPARENTHESIS"]) THEN
            new_tokens.APPEND(("OP", "*"))
        ENDIF
    ENDFOR

    // Perform the last addition to the new_tokens array without checking the next as would be invalid
    new_tokens.APPEND(tokens[length, 1], tokens[length, 2])  
    RETURN new_tokens
ENDSUBROUTINE

SUBROUTINE validateTokens(tokens)
    // Checking if the order of the tokens array is valid
    // Initialise variables
    parenthesis_balance ← 0 
    prev_token ← NULL
    prev_value ← NULL
    variable_found ← FALSE
    variable ← NULL

    // FOR loop contains every validation check to be ran on every token
    FOR index ← 1 TO LEN(tokens) 
        token ← tokens[index, 1]
        value ← tokens[index, 2]

        // Only sets prev_token if one exists
        IF index > 1 THEN
            prev_token ← tokens[index - 1, 1]
            prev_value ← tokens[index - 1, 2]
        ENDIF

        // Only sets next_token if one exists
        IF index < LEN(tokens) THEN
            next_token ← tokens[index + 1, 1]
        ELSE
            next_token ← NULL
        ENDIF

        // Checking for parenthesis balance
        IF token = "LEFTPARENTHESIS" THEN
            parenthesis_balance ← parenthesis_balance + 1
        ELSE IF token = "RIGHTPARENTHESIS" THEN
            parenthesis_balance ← parenthesis_balance - 1
            IF parenthesis_balance < 0 THEN
                // Can end early here as we know ()) is invalid without checking any others
                RETURN FALSE
            ENDIF
        ENDIF

        // Checking for double operators, only allows if the second is a negative
        IF prev_token = "OP" AND token = "OP" AND value <> "-" THEN
            RETURN FALSE
        ENDIF

        // Checking if a trigonometric function is followed by a parenthesis
        IF token = "FUNCTION" AND next_token <> "LEFTPARENTHESIS" THEN
            RETURN FALSE
        ENDIF

        // Checks to see if there is more than one variable attempting to be created
        // If not sets the variable as the one to check
        IF token = "NAME" THEN
            IF variable_found = FALSE THEN
                variable ← value
                variable_found ← TRUE
            ELSE IF value <> variable THEN
                RETURN FALSE
            ENDIF
        ENDIF
    ENDFOR

    // Final check if the last token is an operator or if the parenthesis balance is incorrect
    IF tokens[LEN(tokens), 1] = "OP" OR parenthesis_balance <> 0 THEN
        RETURN FALSE
    ENDIF

    RETURN TRUE
ENDSUBROUTINE

SUBROUTINE parseExpression(expression)
    // Brings together the three subroutines involved in parsing the expression

    //Tokenise the expression
    tokens ← tokenize(expression)

    // If tokenizing is unsuccessful then return NULL
    IF tokens = NULL THEN
        RETURN NULL
    ENDIF

    // Add multiplication that may have been implicit ready for validation
    tokens ← insertImplicitMultiplication(tokens)

    // If validation returns FALSE then return NULL
    IF NOT validateTokens(tokens) THEN
        RETURN NULL
    ENDIF

    // If all subroutines have returned a valid expression then return the token array
    RETURN tokens
ENDSUBROUTINE
