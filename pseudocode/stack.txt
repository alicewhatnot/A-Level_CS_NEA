CLASS Stack
    SUBROUTINE Initialise(max_size)
        // Create fixed-size array for stack items
        items ← array of size max_size filled with NONE

        // Index of the next free position (top of stack)
        top_index ← 0

        // Maximum size of the stack
        max_size ← max_size
    ENDSUBROUTINE

    SUBROUTINE push(value)
        // Adds a value to the top of the stack

        IF top_index = max_size THEN
            OUTPUT "Stack Overflow"
            RETURN
        ENDIF

        items[top_index] ← value
        top_index ← top_index + 1
    ENDSUBROUTINE

    SUBROUTINE pop()
        // Removes and returns the top value from the stack

        IF top_index = 0 THEN
            OUTPUT "Stack Underflow"
            RETURN NONE
        ENDIF

        top_index ← top_index - 1
        value ← items[top_index]
        items[top_index] ← NONE

        RETURN value
    ENDSUBROUTINE

    SUBROUTINE top()
        // Returns the top value without removing it

        IF top_index = 0 THEN
            RETURN NONE
        ENDIF

        RETURN items[top_index - 1]
    ENDSUBROUTINE

    SUBROUTINE empty()
        // Returns TRUE if the stack is empty
        RETURN top_index = 0
    ENDSUBROUTINE

    SUBROUTINE size()
        // Returns the number of items in the stack
        RETURN top_index
    ENDSUBROUTINE
ENDCLASS
