CLASS STACK
    ATTRIBUTES
        PRIVATE items : LIST
        PRIVATE top_index : INTEGER
        PRIVAVTE max_size : INTEGER
    ENDATTRIBUTES
    
    CONSTRUCTOR (max_size)
        // Create fixed-size array for stack items
        items ← EMPTY LIST

        // Index of the next free position (top of stack)
        top_index ← 0

        // Maximum size of the stack
        max_size ← max_size
    ENDCONSTRUCTOR

    PUBLIC METHOD PUSH(value)
        // Adds a value to the top of the stack

        IF top_index = max_size THEN
            OUTPUT "Stack Overflow"
            RETURN
        ENDIF

        items[top_index] ← value
        top_index ← top_index + 1
    ENDMETHOD

    PUBLIC METHOD POP()
        // Removes and returns the top value from the stack

        IF top_index = 0 THEN
            RETURN NONE
        ENDIF

        top_index ← top_index - 1
        value ← items[top_index]
        items[top_index] ← NONE

        RETURN value
    ENDMETHOD

    PUBLIC METHOD TOP()
        // Returns the top value without removing it

        IF top_index = 0 THEN
            RETURN NONE
        ENDIF

        RETURN items[top_index]
    ENDMETHOD

    PUBLIC METHOD EMPTY()
        // Returns TRUE if the stack is empty
        RETURN top_index = 0
    ENDMETHOD

    PUBLIC METHOD SIZE()
        // Returns the number of items in the stack
        RETURN top_index
    ENDMETHOD
ENDCLASS
