CLASS TRANSFORMATION
    // Transformation object created at runtime
    // Stores all information required to apply a transformation

    ATTRIBUTES
        PRIVATE type : STRING
        PRIVATE value : FLOAT
        PRIVATE axis : STRING
    ENDATTRIBUTES
    
    CONSTRUCTOR (transform_type, transform_value, transform_axis)
        // Type of transformation: "shift", "stretch", "reflect", "differentiate", etc.
        type ← transform_type

        // Numeric value used for shift or stretch
        value ← transform_value

        // Axis the transformation applies to ("x" or "y")
        axis ← transform_axis
    ENDCONSTRUCTOR

    PUBLIC METHOD GET_VALUE()
        // Returns the transformation value
        RETURN value
    ENDMETHOD

    PUBLIC METHOD GET_TYPE()
        // Returns the transformation type
        RETURN type
    ENDMETHOD

    PUBLIC METHOD GET_AXIS()
        // Returns the transformation axis
        RETURN axis
    ENDMETHOD
ENDCLASS
